from collections import Counter, defaultdict
import hashlib
import json
import math
import os
import sys

rules = sys.stdin.buffer.read()
directory = os.path.normcase(os.path.abspath(os.getcwd()))
allow = {os.path.join(directory, n) for n in ("prompt.md","input.json")}
opened = []
denied = []
def guard(event,args):
    if event == "open":
        target = os.path.normcase(os.path.abspath(os.fsdecode(args[0]))) if isinstance(args[0],(str,bytes)) else repr(args[0])
        flags = args[2] if len(args)>2 and args[2] is not None else 0
        if target not in allow or flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC):
            denied.append(event)
            raise PermissionError("read limited to prompt and assigned input")
        opened.append(os.path.basename(target))
    elif event in ("os.listdir","os.scandir","os.chdir","os.system","os.startfile","sqlite3.connect","subprocess.Popen") or event.startswith(("socket.","ctypes.","urllib.","winreg.","os.exec","os.spawn")):
        denied.append(event)
        raise PermissionError("external retrieval is disabled")

sys.addaudithook(guard)
for probe in (lambda:open("../outside-scope.probe","rb"),lambda:list(os.scandir(".."))):
    try:
        probe()
    except PermissionError:
        continue
    raise RuntimeError("external retrieval probe unexpectedly succeeded")
with open("prompt.md","rb") as stream:
    prompt = stream.read()
with open("input.json","rb") as stream:
    data = stream.read()
scope={"__name__":"rules"}
exec(compile(rules,"<frozen recognition rules>","exec"),scope)
report=scope["predict"](json.loads(data),hashlib.sha256(prompt).hexdigest().upper())
report["isolation"]={"opened":opened,"denied_probes":denied,
                     "input_sha256":hashlib.sha256(data).hexdigest().upper(),
                     "rules_sha256":hashlib.sha256(rules).hexdigest().upper(),
                     "knowledge_base_entries_visible":0,"retrieval_tools":[],
                     "interpreter":sys.version.split()[0]}
print(json.dumps(report,ensure_ascii=True,allow_nan=False,separators=(",",":")))
